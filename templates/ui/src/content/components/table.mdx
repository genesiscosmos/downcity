# Table

## Usage

```tsx
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@downcity/ui";

export function Example() {
  return <Table><TableHeader><TableRow><TableHead>Name</TableHead></TableRow></TableHeader><TableBody><TableRow><TableCell>Button</TableCell></TableRow></TableBody></Table>;
}
```
