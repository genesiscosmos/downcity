# Combobox

在较多预定义选项中搜索并选择，不用于自由文本搜索。

## Preview

<ComponentDemo component_id="combobox" />

## Usage

```tsx
import { Combobox, ComboboxContent, ComboboxInput, ComboboxInputGroup, ComboboxItem, ComboboxList } from "@downcity/ui";

const options = [{ id: "balanced", label: "Balanced" }];

export function Example() {
  return <Combobox items={options}><ComboboxInputGroup><ComboboxInput aria-label="Style" /></ComboboxInputGroup><ComboboxContent><ComboboxList>{(option) => <ComboboxItem value={option}>{option.label}</ComboboxItem>}</ComboboxList></ComboboxContent></Combobox>;
}
```
